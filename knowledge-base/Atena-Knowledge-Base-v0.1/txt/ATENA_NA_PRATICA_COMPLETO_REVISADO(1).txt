Como usar este livro
A abordagem é a mesma de uma mentoria técnica em chat: fazemos uma pequena parte, entendemos o que
ela representa, compilamos, testamos, diagnosticamos, registramos no Git e só então avançamos. Não existe
“copiar 3 mil linhas e rezar”. Cada marco cria algo executável.
   Ambiente principal: Linux Mint, com Geany como editor leve e terminal como ferramenta de build e
    diagnóstico.
   Linguagem do Core: C17. C++ aparece apenas onde bibliotecas externas exigirem ou onde o projeto
    decidir conscientemente.
   Build: CMake. Testes: CTest + assert no início, evoluindo para sanitizers, integração e fuzzing.
   IA local: llama.cpp/llama-server como runtime de referência; Ollama e APIs remotas entram como providers
    substituíveis.
   Persistência: SQLite. RAG leve: FTS5. Transporte HTTP: libcurl. JSON: json-c.
   Regra de segurança: texto gerado pelo modelo nunca vira shell arbitrário. Toda ação passa por Tool
    Registry, schema, política, whitelist e confirmação quando necessário.
Correção importante da auditoria — Algumas críticas ao PDF anterior estavam certas; outras precisavam
de refinamento. Exemplo: ProtectHome=read-only não impede leitura do HOME, apenas escrita. O problema
real é que ele permite leitura ampla demais. Nesta versão usamos permissões granulares e política de arquivos
aprovados. Também corrigimos a validação de endpoint com parser de URL, pois comparar strings com dois
strncmp negados não é uma validação correta.

    #include "atena/request.h"
    #include <string.h>

    #define ATENA_MAX_USER_TEXT_BYTES (64u * 1024u)

    AtenaStatus atena_request_validate(const AtenaRequest *r) {
        if (!r || !r->session_id || !r->user_text) return ATENA_ERR_INVALID;
        if (r->session_id[0] == '\0' || r->user_text[0] == '\0') return ATENA_ERR_INVALID;
        if (strlen(r->user_text) > ATENA_MAX_USER_TEXT_BYTES) return ATENA_ERR_INVALID;
        if (r->privacy < ATENA_PRIVACY_PUBLIC || r->privacy > ATENA_PRIVACY_SECRET)
            return ATENA_ERR_INVALID;
        if (r->allowed_tool_count > ATENA_MAX_ALLOWED_TOOLS) return ATENA_ERR_INVALID;
        return ATENA_OK;
    }

    bool atena_request_tool_allowed(const AtenaRequest *r, const char *name) {
        if (!r || !name || !r->allow_tools) return false;
        for (size_t i = 0; i < r->allowed_tool_count; ++i) {
            if (r->allowed_tools[i] && strcmp(r->allowed_tools[i], name) == 0) return true;
        }
        return false;
    }

Falha arquitetural corrigida — A whitelist de tools agora é parte explícita do request. O Agent Engine não
pode executar uma tool apenas porque o JSON tem schema válido.

    #include "atena/policy.h"

    AtenaStatus atena_policy_evaluate(const AtenaRequest *r, AtenaPolicyDecision *out) {
        if (!r || !out) return ATENA_ERR_INVALID;
        out->may_use_network = r->allow_network;
        out->may_use_tools = r->allow_tools;
        switch (r->privacy) {
            case ATENA_PRIVACY_PUBLIC:
            case ATENA_PRIVACY_NORMAL:
                out->may_use_remote_provider = r->allow_network;
                break;
            case ATENA_PRIVACY_SENSITIVE:
            case ATENA_PRIVACY_PRIVATE:
            case ATENA_PRIVACY_SECRET:
                out->may_use_remote_provider = false;
                break;
            default:
                return ATENA_ERR_INVALID;
        }
        return ATENA_OK;
    }

Regra — Falha fechada: um valor desconhecido de privacidade não libera a rede. Política vem antes do router
de qualidade/custo.

Parte 10 - Endpoint seguro: não confie em strings
O provider local deve aceitar apenas loopback quando configurado como LOOPBACK_ONLY. Não use testes
manuais frágeis de prefixo. Parseie a URL com a API de URL da libcurl e verifique scheme, host, porta e
ausência de credenciais.
C

    #include <curl/curl.h>
    #include <string.h>

    static bool is_loopback_host(const char *host) {
        return host && (
            strcmp(host, "127.0.0.1") == 0 ||
            strcmp(host, "localhost") == 0 ||
            strcmp(host, "::1") == 0
        );
    }

    AtenaStatus atena_validate_loopback_endpoint(const char *endpoint) {
        if (!endpoint) return ATENA_ERR_INVALID;
        CURLU *u = curl_url();
        if (!u) return ATENA_ERR_NOMEM;
        CURLUcode rc = curl_url_set(u, CURLUPART_URL, endpoint, 0);
        if (rc != CURLUE_OK) { curl_url_cleanup(u); return ATENA_ERR_INVALID; }

       char *scheme=NULL, *host=NULL, *user=NULL, *password=NULL;
       rc = curl_url_get(u, CURLUPART_SCHEME, &scheme, 0);
       if (rc != CURLUE_OK || strcmp(scheme, "http") != 0) goto invalid;
       rc = curl_url_get(u, CURLUPART_HOST, &host, 0);
       if (rc != CURLUE_OK || !is_loopback_host(host)) goto invalid;
       if (curl_url_get(u, CURLUPART_USER, &user, 0) == CURLUE_OK) goto invalid;
       if (curl_url_get(u, CURLUPART_PASSWORD, &password, 0) == CURLUE_OK) goto invalid;

        curl_free(scheme); curl_free(host); curl_url_cleanup(u);
        return ATENA_OK;
    invalid:
        curl_free(scheme); curl_free(host); curl_free(user); curl_free(password);
        curl_url_cleanup(u);
        return ATENA_ERR_INVALID;
    }

Por que a correção sugerida na auditoria não serve — A condição com dois !strncmp unidos por &&
exigiria que a mesma string começasse simultaneamente com “127.0.0.1” e “localhost”. Além disso, prefixo
não valida URL, porta, esquema nem userinfo.

14.1 Espaço livre antes do download
C

     #include <sys/statvfs.h>
     #include <stdint.h>

     AtenaStatus ensure_free_space(const char *path, uint64_t download_bytes) {
         struct statvfs s;
         if (!path || statvfs(path, &s) != 0) return ATENA_ERR_IO;
         uint64_t free_bytes = (uint64_t)s.f_bavail * (uint64_t)s.f_frsize;
         if (download_bytes > UINT64_MAX / 2) return ATENA_ERR_INVALID;
         uint64_t required = download_bytes * 2; /* arquivo + margem/cache/temporários */
         return free_bytes >= required ? ATENA_OK : ATENA_ERR_IO;
     }

O valor 2x é política conservadora, não lei física. O manager pode parametrizar margem absoluta e relativa. O

Parte 15 - Hardware Profiler e compatibilidade
A Atena precisa saber onde está rodando. “2 GB de RAM” não é suficiente: devemos medir memória disponível,
swap, arquitetura, threads, disco e, quando possível, GPU/backends.
C

    #include <sys/sysinfo.h>
    #include <unistd.h>

    typedef struct {
        unsigned long long ram_total;
        unsigned long long ram_free;
        long cpu_threads;
    } AtenaHardwareProfile;

    AtenaStatus atena_hw_profile(AtenaHardwareProfile *out) {
        if (!out) return ATENA_ERR_INVALID;
        struct sysinfo si;
        if (sysinfo(&si) != 0) return ATENA_ERR_IO;
        out->ram_total = (unsigned long long)si.totalram * si.mem_unit;
        out->ram_free = (unsigned long long)si.freeram * si.mem_unit;
        out->cpu_threads = sysconf(_SC_NPROCESSORS_ONLN);
        return ATENA_OK;
    }

Em Linux moderno, “free RAM” isolada é enganosa por causa de page cache. Uma versão avançada lê
/proc/meminfo e usa MemAvailable. O benchmark real deve superar estimativas teóricas.

Conclusão - o que este livro entrega
Atena não termina quando o primeiro modelo responde. Ela termina uma primeira geração quando possui uma
plataforma própria: contratos, política, providers, memória, conhecimento, tools, agentes, auditoria, interface e
integração ao Linux. O modelo continua sendo uma peça substituível. Essa é a diferença entre construir um
chatbot e construir uma capacidade nativa de sistema operacional.
A melhor forma de usar este livro é implementar um marco por vez e manter uma branch “main” sempre
compilável. Quando um capítulo divergir da versão atual de uma biblioteca externa, use o método de pesquisa
descrito aqui, atualize somente o adapter correspondente e registre a decisão. A arquitetura da Atena foi
desenhada precisamente para que o mundo externo possa mudar sem obrigar o Core a se desfazer.
